  add_test(NAME usb_adapter COMMAND usb_adapter_tests)
  add_executable(fake_pico tests/fake_pico.c)
  target_link_libraries(fake_pico PRIVATE devbox_core)
  if(DEVBOX_LWIP_PATH)
    file(GLOB LWIP_SOURCES "${DEVBOX_LWIP_PATH}/src/core/*.c"

    add_executable(network_tests tests/network_tests.c ${LWIP_SOURCES}

    target_include_directories(network_tests PRIVATE tests/mocks include "${DEVBOX_LWIP_PATH}/src/include")

set(PICO_BOARD pico2_w CACHE STRING "Supported board")
if(NOT PICO_BOARD STREQUAL "pico2_w")
  message(FATAL_ERROR "This provisioning layout is verified only for pico2_w")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(picousb C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(picousb src/main.c src/core.c src/websocket.c src/network.c src/usb.c)
target_include_directories(picousb PRIVATE include)
target_link_libraries(picousb pico_stdlib pico_cyw43_arch_lwip_poll pico_lwip_mdns pico_mbedtls tinyusb_device tinyusb_board)
if(PICOUSB_DIAGNOSTICS)
  target_compile_definitions(picousb PRIVATE PICOUSB_DIAGNOSTICS=1)
  target_link_libraries(picousb hardware_watchdog)
endif()
target_compile_options(picousb PRIVATE -Wall -Wextra -Wno-misleading-indentation)
target_compile_definitions(picousb PRIVATE PICO_STACK_SIZE=16384 PICO_USE_STACK_GUARDS=1)
# Preserve the full physical flash size for runtime addressing, but shrink the
# linker FLASH region. Both the linker and post-build check enforce the boundary.
file(READ "${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2350/memmap_default.ld" MEMMAP)
string(REPLACE "INCLUDE \"pico_flash_region.ld\"" "FLASH(rx) : ORIGIN = 0x10000000, LENGTH = 4190208" MEMMAP "${MEMMAP}")
# Network callbacks nest several bounded frame buffers. Reserve 16 KiB in main
# SRAM rather than overflowing the SDK's 2 KiB stack in the 4 KiB scratch bank.
string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 512k"
  "RAM(rwx) : ORIGIN = 0x20000000, LENGTH = 496k\n    NET_STACK(rwx) : ORIGIN = 0x2007c000, LENGTH = 16k" MEMMAP "${MEMMAP}")
string(REPLACE "KEEP(*(.stack*))\n    } > SCRATCH_Y"
  "KEEP(*(.stack*))\n    } > NET_STACK" MEMMAP "${MEMMAP}")
string(REPLACE "__StackTop = ORIGIN(SCRATCH_Y) + LENGTH(SCRATCH_Y);"
  "__StackTop = ORIGIN(NET_STACK) + LENGTH(NET_STACK);" MEMMAP "${MEMMAP}")
string(APPEND MEMMAP "\nASSERT(__StackBottom == 0x2007c000, \"network stack reservation missing\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reserved_flash.ld" "${MEMMAP}")
pico_set_linker_script(picousb "${CMAKE_CURRENT_BINARY_DIR}/reserved_flash.ld")
pico_enable_stdio_uart(picousb 1)
pico_enable_stdio_usb(picousb 0)
pico_add_extra_outputs(picousb)
